// ===== LOGIN (FINAL WORKING VERSION) ===== function handleLogin(e) { e.preventDefault(); const email = document.getElementById("login-email").value.trim(); const password = document.getElementById("login-password").value.trim(); const err = document.getElementById("login-error"); if (err) err.textContent = ""; // Reject personal domains if (!isProfessionalEmail(email)) { err.textContent = "Please login using a company/work email only."; return; } // Check user exists const user = findUser(email); if (!user) { err.textContent = "This email is not registered."; return; } // Password match check if (user.password !== password) { err.textContent = "Incorrect password."; return; } // Save logged-in session sessionStorage.setItem(SESSION_KEY, email); // Redirect to dashboard window.location.href = "dashboard.html"; }
